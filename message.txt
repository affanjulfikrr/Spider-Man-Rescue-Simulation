   SPIDER-MAN RESCUE: INTERACTIVE URBAN RESCUE SIMULATION
   =========================================================================
   C++ / OpenGL / GLUT -- single-file implementation
   Computer Graphics Lab -- Final Showcase Project

   A citizen stands on a rooftop edge while a countdown runs. Press 'S'
   any time before or during the fall to send Spider-Man homing in on a
   web to make the rescue -- the swing continuously re-aims at the
   citizen's live position, so a catch is possible from anywhere, at any
   timing. Missing the window resolves to a clear, non-graphic MISSION

// Plots exactly one pixel -- the base primitive every line/circle
// algorithm below is ultimately built from, kept as its own function so
// ddaLine/bresenhamLine/midpointCircle read as "plot these particular
// pixels" rather than repeating glBegin(GL_POINTS)/glVertex2i/glEnd
// triplets inline at every call site.
void setPixel(int x, int y) {
    glBegin(GL_POINTS);
        glVertex2i(x, y);
    glEnd();
}

// DDA line -- used for window mullions, lane markings, thin guide lines
void ddaLine(double x0, double y0, double x1, double y1) {
    double dx = x1 - x0, dy = y1 - y0;
    int steps = (int) max(fabs(dx), fabs(dy));
    if (steps == 0) { setPixel((int)x0, (int)y0); return; }
    double xInc = dx / steps, yInc = dy / steps;
    double x = x0, y = y0;
    glBegin(GL_POINTS);
    for (int i = 0; i <= steps; ++i) {
        glVertex2i((int)round(x), (int)round(y));
        x += xInc; y += yInc;
    }
    glEnd();
}

// Bresenham line -- used for figure limbs, web strand accents, edges
void bresenhamLine(int x0, int y0, int x1, int y1) {
    int dx = abs(x1 - x0), dy = abs(y1 - y0);
    int sx = (x0 < x1) ? 1 : -1;
    int sy = (y0 < y1) ? 1 : -1;
    int err = dx - dy;
    glBegin(GL_POINTS);
    while (true) {
        glVertex2i(x0, y0);
        if (x0 == x1 && y0 == y1) break;
        int e2 = 2 * err;
        if (e2 > -dy) { err -= dy; x0 += sx; }
        if (e2 <  dx) { err += dx; y0 += sy; }
    }
    glEnd();
}

// Midpoint Circle algorithm -- heads, wheels, sun/moon, stars, canopies
void midpointCircle(double xcF, double ycF, double radius, bool filled) {
    int xc = (int)xcF, yc = (int)ycF;
    int x = 0, y = (int)radius;
    int d = 1 - (int)radius;

    auto plotOctants = [&](int px, int py) {
        if (!filled) {
            glBegin(GL_POINTS);
                glVertex2i(xc+px, yc+py); glVertex2i(xc-px, yc+py);
                glVertex2i(xc+px, yc-py); glVertex2i(xc-px, yc-py);
                glVertex2i(xc+py, yc+px); glVertex2i(xc-py, yc+px);
                glVertex2i(xc+py, yc-px); glVertex2i(xc-py, yc-px);
            glEnd();
        } else {
            glBegin(GL_LINES);
                glVertex2i(xc-px, yc+py); glVertex2i(xc+px, yc+py);
                glVertex2i(xc-px, yc-py); glVertex2i(xc+px, yc-py);
                glVertex2i(xc-py, yc+px); glVertex2i(xc+py, yc+px);
                glVertex2i(xc-py, yc-px); glVertex2i(xc+py, yc-px);
            glEnd();
        }
    };

    plotOctants(x, y);
    while (x < y) {
        x++;
        if (d < 0) d += 2*x + 1;
        else { y--; d += 2*(x - y) + 1; }
        plotOctants(x, y);
    }
}

// Cubic Bezier -- used for the sagging web line during the swing
void bezierCurve(Point2D p0, Point2D p1, Point2D p2, Point2D p3, int segments) {
    glBegin(GL_LINE_STRIP);
    for (int i = 0; i <= segments; ++i) {
        double t = (double)i / segments, u = 1.0 - t;
        double bx = u*u*u*p0.x + 3*u*u*t*p1.x + 3*u*t*t*p2.x + t*t*t*p3.x;
        double by = u*u*u*p0.y + 3*u*u*t*p1.y + 3*u*t*t*p2.y + t*t*t*p3.y;
        glVertex2f((float)bx, (float)by);
    }
    glEnd();
}

// -------------------------------------------------------------------------
// Cohen-Sutherland line clipping
// -------------------------------------------------------------------------
// Classic region-code line-clipping algorithm, applied against the window
// viewport -- used to keep the web strand and debug grid correctly
// bounded even under extreme window resizes.
//
// Region codes (outcodes), one bit per boundary:
//     1001 | 1000 | 1010      bit 0 (LEFT)   : x < xmin
//     -----+------+-----      bit 1 (RIGHT)  : x > xmax
//     0001 | 0000 | 0010      bit 2 (BOTTOM) : y < ymin
//     -----+------+-----      bit 3 (TOP)    : y > ymax
//     0101 | 0100 | 0110
const int CLIP_INSIDE = 0;
const int CLIP_LEFT   = 1;
const int CLIP_RIGHT  = 2;
const int CLIP_BOTTOM = 4;
const int CLIP_TOP    = 8;

int computeOutCode(double x, double y, double xmin, double ymin, double xmax, double ymax) {
    int code = CLIP_INSIDE;
    if      (x < xmin) code |= CLIP_LEFT;
    else if (x > xmax) code |= CLIP_RIGHT;
    if      (y < ymin) code |= CLIP_BOTTOM;
    else if (y > ymax) code |= CLIP_TOP;
    return code;
}

// Clips the segment (x0,y0)-(x1,y1) against [xmin,xmax] x [ymin,ymax].
// Returns true and overwrites the endpoints if any part of the segment is
// visible; returns false if the whole segment lies outside the window.
bool cohenSutherlandClip(double& x0, double& y0, double& x1, double& y1,
                          double xmin, double ymin, double xmax, double ymax) {
    int outcode0 = computeOutCode(x0, y0, xmin, ymin, xmax, ymax);
    int outcode1 = computeOutCode(x1, y1, xmin, ymin, xmax, ymax);
    bool accept = false;

    while (true) {
        if (!(outcode0 | outcode1)) {
            // Both endpoints inside the clip window.
            accept = true;
            break;
        } else if (outcode0 & outcode1) {
            // Both endpoints share an outside region -> segment fully rejected.
            break;
        } else {
            // At least one endpoint is outside; pick it and clip against
            // whichever boundary it violates.
            double x = 0.0, y = 0.0;
            int outcodeOut = outcode0 ? outcode0 : outcode1;

            if (outcodeOut & CLIP_TOP) {
                x = x0 + (x1 - x0) * (ymax - y0) / (y1 - y0);
                y = ymax;
            } else if (outcodeOut & CLIP_BOTTOM) {
                x = x0 + (x1 - x0) * (ymin - y0) / (y1 - y0);
                y = ymin;
            } else if (outcodeOut & CLIP_RIGHT) {
                y = y0 + (y1 - y0) * (xmax - x0) / (x1 - x0);
                x = xmax;
            } else { // CLIP_LEFT
                y = y0 + (y1 - y0) * (xmin - x0) / (x1 - x0);
                x = xmin;
            }

            if (outcodeOut == outcode0) {
                x0 = x; y0 = y;
                outcode0 = computeOutCode(x0, y0, xmin, ymin, xmax, ymax);
            } else {
                x1 = x; y1 = y;
                outcode1 = computeOutCode(x1, y1, xmin, ymin, xmax, ymax);
            }
        }
    }
    return accept;
}

// Draws a Bresenham-rasterized segment only after clipping it to the
// window bounds -- used for the web strand so a swing that overshoots the
// window during a resize never produces off-screen artifacts.
void drawClippedLine(double x0, double y0, double x1, double y1) {
    double cx0 = x0, cy0 = y0, cx1 = x1, cy1 = y1;
    if (cohenSutherlandClip(cx0, cy0, cx1, cy1, 0.0, 0.0, (double)WINDOW_WIDTH, (double)WINDOW_HEIGHT)) {
        bresenhamLine((int)cx0, (int)cy0, (int)cx1, (int)cy1);
    }
}

// filled=true draws a solid GL_QUADS rectangle; filled=false draws just
// its GL_LINE_LOOP outline. Used for everything axis-aligned in the
// scene: torsos, building bodies, vehicle bodies, HUD panels.
void drawRectangle(double x, double y, double w, double h, bool filled) {
    glBegin(filled ? GL_QUADS : GL_LINE_LOOP);
        glVertex2f((float)x, (float)y);
        glVertex2f((float)(x+w), (float)y);
        glVertex2f((float)(x+w), (float)(y+h));
        glVertex2f((float)x, (float)(y+h));
    glEnd();
}

// Thin wrapper around GLUT's bitmap font rendering, bundling the color
// set + raster position + character loop that would otherwise be repeated
// at every text call site in the HUD and banners.
void drawText(double x, double y, const string& text, void* font, double r, double g, double b) {
    glColor3f((float)r, (float)g, (float)b);
    glRasterPos2f((float)x, (float)y);
    for (char c : text) glutBitmapCharacter(font, c);
}

// Measures the pixel width of the string in the given bitmap font (via
// glutBitmapWidth) and calls drawText with an x offset that centers it on
// cx -- used by every banner and HUD headline.
void drawCenteredText(double cx, double y, const string& text, void* font, double r, double g, double b) {
    int w = 0;
    for (char c : text) w += glutBitmapWidth(font, c);
    drawText(cx - w / 2.0, y, text, font, r, g, b);
}

// =====================================================================
// 2D TRANSFORMATIONS
// =====================================================================
// Three explicit, reusable transform functions -- every place in this
// program that moves, resizes, or rotates a point goes through one of
// these rather than inlining the arithmetic at the call site. This keeps
// the transform math itself in exactly one place (easy to verify/quote in
// a report) and makes every call site read as "translate/scale/rotate
// this point" rather than a wall of unlabeled add/multiply/trig.

// Translation: simple vector addition. dx/dy can be any signed offset;
// most call sites derive them from a velocity times FIXED_DT.
Point2D translatePoint(const Point2D& p, double dx, double dy) {
    return { p.x + dx, p.y + dy };
}

// Scaling about an arbitrary pivot (not just the origin): shift the point
// into pivot-relative coordinates, scale each axis independently by
// sx/sy, then shift back. sx == sy gives uniform scaling; sy = -1 (as
// used by drawWetReflections) gives a vertical mirror about the pivot.
Point2D scalePoint(const Point2D& p, const Point2D& pivot, double sx, double sy) {
    Point2D rel = { p.x - pivot.x, p.y - pivot.y };
    rel.x *= sx; rel.y *= sy;
    return { rel.x + pivot.x, rel.y + pivot.y };
}

// Rotation about an arbitrary pivot, angle given in degrees for
// readability at call sites (converted to radians internally). Standard
// 2x2 rotation matrix applied to pivot-relative coordinates. This is the
// function underlying Spider-Man's entire swing motion (updateSpiderSwing)
// and the pedestrians' leg-swing walk cycle (drawPedestrians).
Point2D rotatePoint(const Point2D& p, const Point2D& pivot, double angleDegrees) {
    double rad = angleDegrees * PI / 180.0;
    double cosA = cos(rad), sinA = sin(rad);
    Point2D rel = { p.x - pivot.x, p.y - pivot.y };
    Point2D rotated = { rel.x*cosA - rel.y*sinA, rel.x*sinA + rel.y*cosA };
    return { rotated.x + pivot.x, rotated.y + pivot.y };
}

// =====================================================================
// LIGHTING (day / night lerp)
// =====================================================================
// Looks up the constant palette for a given mode -- the only two concrete
// lighting states in the program; every on-screen color you actually see
// is either one of these two structs directly, or lerpPalette's
// interpolation between them.
ModePalette paletteForMode(LightingMode mode) {
    return mode == LightingMode::DAY ? PALETTE_DAY : PALETTE_NIGHT;
}

ModePalette lerpPalette(const ModePalette& a, const ModePalette& b, double t) {
    t = max(0.0, min(1.0, t));
    ModePalette r;
    r.skyTopR = a.skyTopR + (b.skyTopR - a.skyTopR)*t;
    r.skyTopG = a.skyTopG + (b.skyTopG - a.skyTopG)*t;
    r.skyTopB = a.skyTopB + (b.skyTopB - a.skyTopB)*t;
    r.skyBotR = a.skyBotR + (b.skyBotR - a.skyBotR)*t;
    r.skyBotG = a.skyBotG + (b.skyBotG - a.skyBotG)*t;
    r.skyBotB = a.skyBotB + (b.skyBotB - a.skyBotB)*t;
    r.buildingShade    = a.buildingShade + (b.buildingShade - a.buildingShade)*t;
    r.windowLitChance  = a.windowLitChance + (b.windowLitChance - a.windowLitChance)*t;
    r.lampOn           = a.lampOn + (b.lampOn - a.lampOn)*t;
    r.sunAlpha  = a.sunAlpha  + (b.sunAlpha  - a.sunAlpha)*t;
    r.moonAlpha = a.moonAlpha + (b.moonAlpha - a.moonAlpha)*t;
    r.sunR = a.sunR + (b.sunR - a.sunR)*t;
    r.sunG = a.sunG + (b.sunG - a.sunG)*t;
    r.sunB = a.sunB + (b.sunB - a.sunB)*t;
    return r;
}

// Advances the day/night crossfade by one fixed step per tick whenever a
// transition is in progress (see requestLightMode), then snaps
// activeLightMode over once the interpolation reaches 1.0 so subsequent
// calls stop re-lerping between two identical palettes.
void updateLighting() {
    if (lightTransition < 1.0) {
        lightTransition = min(1.0, lightTransition + LIGHT_TRANSITION_STEP);
        currentPalette = lerpPalette(paletteForMode(activeLightMode), paletteForMode(nextLightMode), lightTransition);
        if (lightTransition >= 1.0) activeLightMode = nextLightMode;
    } else {
        currentPalette = paletteForMode(activeLightMode);
    }
}

// Called by the 'D'/'N' keys: sets the target mode and resets
// lightTransition to 0 so updateLighting starts a fresh crossfade toward
// it. A no-op if we're already settled on the requested mode.
void requestLightMode(LightingMode mode) {
    if (mode == activeLightMode && lightTransition >= 1.0) return;
    nextLightMode = mode;
    lightTransition = 0.0;
    logEvent(mode == LightingMode::DAY ? "Lighting transitioning to DAY." : "Lighting transitioning to NIGHT.");
}

// =====================================================================

// Nine buildings laid out left to right with hand-tuned widths/heights so
// the skyline reads as varied rather than a uniform repeated block. Colors
// are muted blue-grays, later modulated per-frame by currentPalette so the
// same base building looks right in both day and night lighting. The last
// two are purely decorative background buildings (no story role) added to
// carry the skyline across the full window width instead of leaving open
// sky past Building 7 -- the game's building indices below only ever
// reference the first seven, so they're unaffected by the addition.
void initBuildings() {
    buildings.clear();
    double widths[9]  = {110, 95, 100, 90, 100, 95, 115, 95, 80};

// Vertical two-color gradient quad for the sky, blended per-channel by
// currentPalette, plus a scattered star field that only becomes visible
// once moonAlpha rises past a small threshold during the night transition.
void drawSky() {
    glBegin(GL_QUADS);
        glColor3f((float)currentPalette.skyTopR, (float)currentPalette.skyTopG, (float)currentPalette.skyTopB);
        glVertex2f(0, WINDOW_HEIGHT); glVertex2f(WINDOW_WIDTH, WINDOW_HEIGHT);
        glColor3f((float)currentPalette.skyBotR, (float)currentPalette.skyBotG, (float)currentPalette.skyBotB);
        glVertex2f(WINDOW_WIDTH, (float)SIDEWALK_Y); glVertex2f(0, (float)SIDEWALK_Y);
    glEnd();

    if (currentPalette.moonAlpha > 0.05) {
        glColor4f(1.0f, 1.0f, 1.0f, (float)currentPalette.moonAlpha);
        for (int i = 0; i < 50; ++i) {
            int sx = (i * 137) % WINDOW_WIDTH;
            int sy = (int)SIDEWALK_Y + 350 + (i * 71) % 300;
            setPixel(sx, sy);
        }
    }
}

// Sun and moon share one small orbit path (a slight ellipse via
// sunMoonAngle) and are cross-faded by sunAlpha/moonAlpha rather than
// being swapped outright, so during the lighting transition both can be
// partially visible for a brief, natural-looking overlap. The sun also
// gets a few progressively larger, progressively fainter glow rings drawn
// underneath it via repeated midpointCircle calls.
void drawSunMoon() {
    double cx = 130.0 + 12.0 * sin(sunMoonAngle);
    double cy = 610.0 + 8.0  * cos(sunMoonAngle);

    if (currentPalette.sunAlpha > 0.02) {
        for (int i = 3; i >= 1; --i) {
            glColor4f((float)currentPalette.sunR, (float)currentPalette.sunG, (float)currentPalette.sunB,
                      (float)(0.08 * i * currentPalette.sunAlpha));
            midpointCircle(cx, cy, 32.0 + i*7, true);
        }
        glColor4f((float)currentPalette.sunR, (float)currentPalette.sunG, (float)currentPalette.sunB, (float)currentPalette.sunAlpha);
        midpointCircle(cx, cy, 32.0, true);
    }
    if (currentPalette.moonAlpha > 0.02) {
        glColor4f(0.92f, 0.92f, 0.96f, (float)currentPalette.moonAlpha);
        midpointCircle(cx, cy, 26.0, true);
        glColor4f(0.05f, 0.06f, 0.14f, (float)currentPalette.moonAlpha);
        midpointCircle(cx + 10, cy + 4, 22.0, true);
    }
}

// Each cloud is three overlapping midpointCircle discs, tinted darker at night.
void drawSingleCloud(const Cloud& c) {
    double shade = 1.0 - 0.45 * currentPalette.lampOn;
    glColor3f((float)shade, (float)shade, (float)shade);
    midpointCircle(c.x, c.y, 18.0*c.scale, true);
    midpointCircle(c.x + 22*c.scale, c.y + 6*c.scale, 22.0*c.scale, true);
    midpointCircle(c.x - 20*c.scale, c.y + 4*c.scale, 16.0*c.scale, true);
}
void drawClouds() { for (auto& c : clouds) drawSingleCloud(c); }

void updateClouds() {

    }
}

// -------------------------------------------------------------------------
// Birds -- small ambient decoration, daytime only. Each bird is a pair of
// Bresenham strokes forming a shallow "V" whose wing angle oscillates with
// wingPhase, giving a simple flap animation with no additional primitives.
// -------------------------------------------------------------------------
void initBirds() {
    birds.clear();
    for (int i = 0; i < 6; ++i) {
        double x = (double)((i * 197) % WINDOW_WIDTH);
        double y = 560.0 + (double)((i * 83) % 100);
        double speed = 30.0 + 8.0 * (i % 3);
        birds.push_back({ x, y, speed, (double)i });
    }
}

void updateBirds() {
    for (auto& b : birds) {
        Point2D moved = translatePoint({b.x, b.y}, b.speed * FIXED_DT, 0.0);
        b.x = moved.x;
        b.wingPhase += 0.25;
        if (b.x > WINDOW_WIDTH + 20) b.x = -20.0;
    }
}

void drawBirds() {
    // Birds tuck away once the streetlamps are lit -- a cheap day/night cue
    // that costs nothing extra since it reuses the existing lampOn value.
    if (currentPalette.lampOn > 0.6) return;
    glColor3f(0.15f, 0.15f, 0.15f);
    for (auto& b : birds) {
        double flap = 3.0 + 2.0 * sin(b.wingPhase);
        bresenhamLine((int)(b.x - 8), (int)(b.y - flap), (int)b.x, (int)b.y);
        bresenhamLine((int)(b.x + 8), (int)(b.y - flap), (int)b.x, (int)b.y);
    }
}

void drawBuildings() {
    for (size_t i = 0; i < buildings.size(); ++i) {
        Building& b = buildings[i];
        double shade = currentPalette.buildingShade;
        glColor3f((float)(b.r*shade*1.6), (float)(b.g*shade*1.6), (float)(b.b*shade*1.6));
        drawRectangle(b.x, SIDEWALK_Y, b.w, b.h, true);

        // window grid: DDA mullion guide lines behind lit/unlit panes
        glColor3f((float)(b.r*0.25), (float)(b.g*0.25), (float)(b.b*0.3));
        int cols = max(1, (int)(b.w / 20));
        int rows = max(1, (int)(b.h / 26));
        for (int r = 0; r < rows; ++r)
            ddaLine(b.x + 4, SIDEWALK_Y + 6 + r*26, b.x + b.w - 4, SIDEWALK_Y + 6 + r*26);
        for (int c = 0; c < cols; ++c)
            ddaLine(b.x + 4 + c*20, SIDEWALK_Y + 4, b.x + 4 + c*20, SIDEWALK_Y + b.h - 4);

        for (int r = 0; r < rows; ++r) {
            for (int c = 0; c < cols; ++c) {
                double wx = b.x + 8 + c*20, wy = SIDEWALK_Y + 12 + r*26;
                double pseudo = fmod((r*37 + c*53 + i*19) * 0.61803, 1.0);
                bool lit = pseudo < currentPalette.windowLitChance;
                if (lit) glColor3f(1.0f, 0.85f, 0.4f);
                else     glColor3f((float)(b.r*0.4), (float)(b.g*0.4), (float)(b.b*0.45));
                drawRectangle(wx, wy, 9, 14, true);
            }
        }

        drawRooftopDetail(b.x, b.w, SIDEWALK_Y + b.h, b.r, b.g, b.b);
        drawAwning(b.x + b.w*0.15, b.w*0.7, SIDEWALK_Y + 2);
    }
}

// Every vehicle just moves right at its own fixed speed and wraps back to
// a staggered off-screen start once it clears the right edge -- simple
// modular motion rather than a full traffic/spacing simulation, which
// would be unnecessary complexity for a background layer.

    for (auto& v : vehicles) { v.x += v.speed * FIXED_DT; if (v.x > WINDOW_WIDTH + 100) v.x = -150; }

void animatePedestrians() {

        p.x += p.speed * FIXED_DT;
        if (p.x > WINDOW_WIDTH - 20) p.x = 20;
        if (p.x < 20) p.x = WINDOW_WIDTH - 20;
        p.walkPhase += fabs(p.speed) * FIXED_DT * 0.35; // drives the leg-swing offset

    }
}

// =====================================================================
// ACTORS: victim and Spider-Man (built from primitives + transforms)
// =====================================================================
// Shared figure-drawing routine for every humanoid (citizen + all four
// pedestrians): legs are filled rectangles shifted sideways by legSwing
// for the walk cycle, torso is a filled rectangle, arms are strokes in a
// darker shirt shade, head is a skin-toned circle under a hair cap.
// Every point goes through scalePoint then rotatePoint about (x,y), so
// rotDeg = 90 produces the lying-down FAILURE pose from the same geometry.
void drawFigure(double x, double y, const Outfit& o, double scale, double rotDeg = 0.0, double legSwing = 0.0) {
    Point2D base = {x, y};
    auto T = [&](double px, double py) {
        return rotatePoint(scalePoint({px, py}, base, scale, scale), base, rotDeg);
    };
    auto fillQuad = [&](double rx, double ry, double rw, double rh) {
        Point2D a=T(rx,ry), b=T(rx+rw,ry), c=T(rx+rw,ry+rh), d=T(rx,ry+rh);
        glBegin(GL_QUADS);
            glVertex2f((float)a.x,(float)a.y); glVertex2f((float)b.x,(float)b.y);
            glVertex2f((float)c.x,(float)c.y); glVertex2f((float)d.x,(float)d.y);
        glEnd();
    };

    // Legs -- filled rectangles, shifted sideways by legSwing (walk cycle)
    glColor3f((float)o.pantsR, (float)o.pantsG, (float)o.pantsB);
    fillQuad(x - 4 + legSwing*0.2, y,      3, 14);
    fillQuad(x + 1 - legSwing*0.2, y,      3, 14);
    glColor3f(0.12f, 0.12f, 0.12f); // shoes
    fillQuad(x - 4 + legSwing*0.2, y - 2,  3, 2);
    fillQuad(x + 1 - legSwing*0.2, y - 2,  3, 2);

    // Torso
    glColor3f((float)o.shirtR, (float)o.shirtG, (float)o.shirtB);
    fillQuad(x - 5, y + 14, 10, 16);

    // Arms -- a darker shade of the shirt color, swinging opposite the legs
    glColor3f((float)(o.shirtR*0.8), (float)(o.shirtG*0.8), (float)(o.shirtB*0.8));
    Point2D armL0 = T(x-5, y+26), armL1 = T(x-8+legSwing, y+16);
    Point2D armR0 = T(x+5, y+26), armR1 = T(x+8-legSwing, y+16);
    bresenhamLine((int)armL0.x,(int)armL0.y,(int)armL1.x,(int)armL1.y);
    bresenhamLine((int)armR0.x,(int)armR0.y,(int)armR1.x,(int)armR1.y);

    // Hair cap, then face on top of it
    Point2D hairPos = T(x, y+34);
    glColor3f((float)o.hairR, (float)o.hairG, (float)o.hairB);
    midpointCircle(hairPos.x, hairPos.y, 5.8*scale, true);
    Point2D facePos = T(x, y+32);
    glColor3f((float)o.skinR, (float)o.skinG, (float)o.skinB);
    midpointCircle(facePos.x, facePos.y, 5.0*scale, true);
}

// A short walk-cycle phase advanced only while the citizen is actually
// walking (the WALK state); kept separate from the pedestrians' phases
// so the citizen's gait doesn't need to share their vector.
double victimWalkPhase = 0.0;

// The citizen is drawFigure() in a fixed outfit. Standing upright through
// WAITING/JUMPING/RESCUE/CARRY/TALK; legs swing only during WALK. On
// FAILURE the citizen is placed resting at ground level and drawn
// rotated 90 degrees (lying down) rather than disappearing -- the
// outcome stays visible and honest without any graphic detail, and
// nothing slides or vanishes off-screen.
void drawVictim() {
    double rot = (state == GameState::FAILURE) ? 90.0 : 0.0;
    double legSwing = (state == GameState::WALK) ? sin(victimWalkPhase) * 4.0 : 0.0;
    if (state != GameState::FAILURE && state != GameState::JUMPING) {
        glColor4f(0.0f, 0.0f, 0.0f, 0.18f);
        midpointCircle(victim.x, victim.y, 6.0, true);
    }
    drawFigure(victim.x, victim.y, victimOutfit, 1.0, rot, legSwing);
}

// Spider-Man is built from the exact same drawFigure() used by every
// pedestrian and the citizen -- same legs/torso/arms/head construction --
// just reskinned as a red-and-blue costume (mask-red "skin", a darker red
// hair-cap shade for the mask crown, red shirt, blue tights) with a fixed
// asymmetric arm/leg pose (legSwing) so he reads as an active, one-arm-
// raised swinger rather than a standing pedestrian. The two-lens eyes,
// web-pattern chest band, belt, emblem, and web-shooter cuff are then
// layered on top at the same coordinates drawFigure uses internally.
void drawSpiderMan() {
    if (!spiderActive) return;
    static const Outfit spidey = { 0.85,0.05,0.08,  0.55,0.03,0.05,  0.85,0.05,0.08,  0.12,0.20,0.55 };
    double x = spider.x, y = spider.y;
    drawFigure(x, y, spidey, 1.0, 0.0, -8.0); // a wide-armed swinging pose, not a standing gait

    // two-lens mask eyes, over the face circle drawFigure already drew
    glColor3f(0.95f, 0.95f, 0.95f);
    midpointCircle(x-3, y+33, 2.2, true);
    midpointCircle(x+3, y+33, 2.2, true);
    glColor3f(0.05f, 0.05f, 0.08f);
    midpointCircle(x-3, y+33, 1.1, true);
    midpointCircle(x+3, y+33, 1.1, true);

    // web-pattern chest band, belt, and emblem, over the torso quad
    glColor3f(0.05f, 0.05f, 0.1f);
    drawRectangle(x-5, y+23, 10, 3, true);
    drawRectangle(x-6, y+14, 12, 3, true); // belt
    glColor3f(0.02f, 0.02f, 0.04f);
    midpointCircle(x, y+22, 3.0, false); // outline "emblem" ring, unfilled circle

    // web-shooter cuff near the raised wrist
    glColor3f(0.1f, 0.1f, 0.12f);
    drawRectangle(x-10, y+14, 4, 3, true);
}

// Web line drawn as a sagging cubic Bezier from the anchor to Spider-Man's
// hand. The curve itself is drawn with the GL line strip for a smooth
// appearance, but each of its straight sub-segments is also independently
// passed through cohenSutherlandClip before being re-rendered as a
// Bresenham stroke slightly offset to the side -- a visible "core strand"
// that demonstrates the clipping algorithm operating on real per-frame
// geometry rather than only the static debug grid.
void drawWeb() {
    if (!showWeb) return;
    Point2D p0 = webAnchor;
    Point2D p3 = { spider.x, spider.y + 22 };
    double midX = (p0.x + p3.x) / 2.0;
    double sag  = 18.0 + 0.05 * fabs(p0.x - p3.x);
    Point2D p1 = { p0.x + (midX - p0.x) * 0.4, p0.y - sag * 0.6 };
    Point2D p2 = { p3.x + (midX - p3.x) * 0.4, p3.y - sag * 0.2 };

    glColor4f(0.9f, 0.9f, 0.9f, 0.85f);
    bezierCurve(p0, p1, p2, p3, 24);

    // Sample the same Bezier at a coarser resolution for the clipped core
    // strand, so the clipping pass has real segment endpoints to work with.
    glColor4f(1.0f, 1.0f, 1.0f, 0.5f);
    const int coreSegments = 8;
    Point2D prev = p0;
    for (int i = 1; i <= coreSegments; ++i) {
        double t = (double)i / coreSegments, u = 1.0 - t;
        Point2D cur = {
            u*u*u*p0.x + 3*u*u*t*p1.x + 3*u*t*t*p2.x + t*t*t*p3.x,
            u*u*u*p0.y + 3*u*u*t*p1.y + 3*u*t*t*p2.y + t*t*t*p3.y
        };
        drawClippedLine(prev.x, prev.y, cur.x, cur.y);
        prev = cur;
    }
}

// =====================================================================
// IMPACT PARTICLES (tasteful, non-graphic failure cue)
// =====================================================================
// This module exists solely to give the FAILURE outcome a clear, felt
// "something happened" moment without depicting injury: 18 small dots are
// scattered outward from the impact point in a full circle, fall briefly
// under half-strength gravity, and fade out over roughly a second. No
// color or motion here is meant to read as blood -- it's a generic,
// abstracted burst of the kind used in many family-friendly games for a
// failure state.
void spawnImpactParticles(double x, double y) {
    particles.clear();
    for (int i = 0; i < 18; ++i) {
        double ang = (double)i / 18.0 * 2.0 * PI;
        double speed = 40.0 + (i % 5) * 12.0;
        particles.push_back({ x, y, cos(ang)*speed, sin(ang)*speed*0.6 + 40.0, 1.0 });
    }
    particlesActive = true;
}

// Advances every particle by one tick and clears particlesActive once all
// of them have faded (life <= 0), so drawImpactParticles/activeParticleCount
// can cheaply skip work once the burst is finished instead of checking
// each particle's lifetime every frame from the outside.
void animateImpactParticles() {
    if (!particlesActive) return;
    bool any = false;
    for (auto& p : particles) {
        Point2D moved = translatePoint({p.x,p.y}, p.vx*FIXED_DT, p.vy*FIXED_DT);
        p.x = moved.x; p.y = moved.y;
        p.vy -= GRAVITY * 0.5 * FIXED_DT;
        p.life -= FIXED_DT * 0.9;
        if (p.life > 0) any = true;
    }
    if (!any) particlesActive = false;
}

// Each particle is a small filled midpointCircle whose alpha is driven
// directly by its remaining life, giving the burst a natural fade-out
// without a separate timer or animation curve.
void drawImpactParticles() {
    if (!particlesActive) return;
    for (auto& p : particles) {
        glColor4f(0.85f, 0.2f, 0.2f, (float)p.life);
        midpointCircle(p.x, p.y, 3.0, true);
    }
}
// Reports how many impact particles are currently live -- used only by
// the diagnostics info panel, so it deliberately doesn't need to be exact
// once particlesActive is false (returns 0 immediately in that case).
int activeParticleCount() { return particlesActive ? (int)particles.size() : 0; }

// =====================================================================
// GAME LOGIC
// =====================================================================

// Homing swing: rather than a fixed-radius, fixed-speed pendulum, the
// swing continuously re-aims itself at the citizen's live position (in
// polar coordinates around the web anchor) every tick, at a fast but
// capped angular/radial speed. This guarantees Spider-Man can intercept
// the citizen from any starting height and at any press timing -- there
// is no "safe window" the player has to time correctly; pressing S at
// the very last instant still results in a catch, just a tighter one.
const double SWING_MAX_ANGULAR_SPEED = 480.0; // degrees/sec
const double SWING_MAX_RADIAL_SPEED  = 720.0; // px/sec

// Sets Spider-Man's initial polar position (angle + radius around the
// anchor) to match where he's actually standing, so the homing swing
// starts exactly from his rooftop position rather than snapping.
void beginRescue() {
    webAnchor = { buildingCenterX(ANCHOR_BUILDING), buildingTopY(ANCHOR_BUILDING) };
    double dx = spider.x - webAnchor.x, dy = spider.y - webAnchor.y;
    swingLen = sqrt(dx*dx + dy*dy);
    swingAngleDeg = atan2(dx, -dy) * 180.0 / PI;
    spiderActive = true;
    showWeb = true;
    state = GameState::RESCUE;
    stateTimer = 0.0;
    logEvent("Spider-Man launched on the rescue web.");
}
void triggerSave() {
    if (state != GameState::WAITING && state != GameState::JUMPING) return;
    logEvent("'S' pressed -- attempting rescue.");
    if (state == GameState::WAITING) { state = GameState::JUMPING; stateTimer = 0.0; victimVel = {0,0}; }
    beginRescue();
}

// Simple radius-threshold collision test between Spider-Man and the
// citizen -- sufficient given both are small stylized figures; a full
// axis-aligned bounding-box test would be overkill and visually no more
// accurate at this figure scale.
bool collisionSpiderVictim() {
    double dx = spider.x - victim.x, dy = spider.y - victim.y;
    return sqrt(dx*dx + dy*dy) < 26.0;
}

// The citizen has "reached the street" once their y-position drops to
// just above the sidewalk band -- checked every JUMPING tick.
bool collisionVictimGround() { return victim.y <= SIDEWALK_Y + 20.0; }

// Semi-implicit Euler integration of free fall: velocity accumulates
// GRAVITY every tick, then position is translated downward by that
// velocity scaled by the timestep. translatePoint takes a negative dy
// since world-space y increases upward in this projection.
void updateVictimFall() {
    victimVel.y += GRAVITY * FIXED_DT;
    Point2D moved = translatePoint(victim, 0.0, -victimVel.y * FIXED_DT);
    victim = moved;
}

// The swing is guided, not a fixed pendulum: every tick, the target angle
// and target web length are recomputed from Spider-Man's anchor point to
// the citizen's CURRENT falling position, and swingAngleDeg/swingLen are
// stepped toward those targets at a bounded angular/radial speed (via
// rotatePoint around the anchor each frame). This is what makes a catch
// possible from anywhere and at any timing -- earlier versions rotated at
// a fixed rate along a fixed-radius arc that only intersected the citizen
// by coincidence; now the web actively reels in and swings to track a
// moving target, the way an aimed rescue actually would.
void updateSpiderSwing() {
    double dx = victim.x - webAnchor.x, dy = victim.y - webAnchor.y;
    double targetAngle = atan2(dx, -dy) * 180.0 / PI;
    double targetLen   = sqrt(dx*dx + dy*dy);

    double angleDiff = targetAngle - swingAngleDeg;
    while (angleDiff > 180.0)  angleDiff -= 360.0;
    while (angleDiff < -180.0) angleDiff += 360.0;
    double maxAngleStep = SWING_MAX_ANGULAR_SPEED * FIXED_DT;
    swingAngleDeg += (fabs(angleDiff) < maxAngleStep) ? angleDiff
                                                        : (angleDiff > 0 ? maxAngleStep : -maxAngleStep);

    double lenDiff = targetLen - swingLen;
    double maxLenStep = SWING_MAX_RADIAL_SPEED * FIXED_DT;
    swingLen += (fabs(lenDiff) < maxLenStep) ? lenDiff : (lenDiff > 0 ? maxLenStep : -maxLenStep);

    Point2D restPos = { webAnchor.x, webAnchor.y - swingLen }; // straight down from anchor
    spider = rotatePoint(restPos, webAnchor, swingAngleDeg);
}

void updateGame() {
    if (isPaused) return;
    tickCounter++;
    stateTimer += FIXED_DT;
    animateCars();
    animatePedestrians();
    updateClouds();
    updateBirds();
    updatePlane();
    animateImpactParticles();
    updateLighting();
    if (shakeTimeLeft > 0.0) shakeTimeLeft -= FIXED_DT;
    sunMoonAngle += 0.01;
    if (sunMoonAngle > 2*PI) sunMoonAngle -= 2*PI;

    switch (state) {
        // ---------------------------------------------------------------
        // WAITING: the citizen stands at the rooftop edge. The countdown
        // is purely a tension device -- pressing S here is just as valid
        // as pressing it during the fall, since triggerSave() handles both.
        // ---------------------------------------------------------------
        case GameState::WAITING:
            countdown -= FIXED_DT;
            if (countdown <= 0.0) { state = GameState::JUMPING; stateTimer = 0.0; victimVel = {0,0}; }
            break;

        // ---------------------------------------------------------------
        // JUMPING: free-fall under gravity (see updateVictimFall). Ends
        // either by a successful 'S' press (which jumps straight into
        // RESCUE via beginRescue/triggerSave) or by the citizen reaching
        // the sidewalk, which resolves to FAILURE with the tasteful
        // particle-burst cue and a brief camera shake -- no gore.
        // ---------------------------------------------------------------
        case GameState::JUMPING:
            updateVictimFall();
            if (collisionVictimGround()) {
                spawnImpactParticles(victim.x, SIDEWALK_Y + 6.0);
                victim.y = SIDEWALK_Y + 6.0; // rest at ground level; drawVictim() lies them down here
                shakeTimeLeft = 0.35; // brief, subtle camera shake -- not gore, just weight
                state = GameState::FAILURE; stateTimer = 0.0;
                logEvent("MISSION FAILED -- citizen was not reached in time.");
            }
            break;

        // ---------------------------------------------------------------
        // RESCUE: Spider-Man swings on the web (updateSpiderSwing, driven
        // by rotatePoint) while the citizen keeps falling under the same
        // gravity model as JUMPING. A successful catch is a simple
        // distance-threshold collision check, not full rectangle overlap
        // -- appropriate given both actors are drawn as small figures.
        // ---------------------------------------------------------------
        case GameState::RESCUE:
            updateSpiderSwing();
            updateVictimFall();
            if (collisionSpiderVictim()) { state = GameState::CARRY; stateTimer = 0.0; }
            break;

        // ---------------------------------------------------------------
        // CARRY: Spider-Man walks the caught citizen (locked to his
        // position with a small offset) toward the landing spot in front
        // of Building 3, at a constant speed via translatePoint. Once
        // within one step of the target, position snaps exactly onto it
        // to avoid an asymptotic "never quite arrives" approach.
        // ---------------------------------------------------------------
        case GameState::CARRY: {
            showWeb = false;
            double tx = buildingCenterX(LAND_BUILDING) - 40.0, ty = SIDEWALK_Y + 30.0;
            double dx = tx - spider.x, dy = ty - spider.y, dist = sqrt(dx*dx+dy*dy);
            double step = 220.0 * FIXED_DT;
            if (dist > step) { Point2D m = translatePoint(spider, dx/dist*step, dy/dist*step); spider = m; }
            else { spider = {tx, ty}; state = GameState::TALK; stateTimer = 0.0; }
            victim = { spider.x + 14.0, spider.y };
            break;
        }

        // ---------------------------------------------------------------
        // TALK: a fixed 2-second pause with a speech-bubble line drawn in
        // drawHUD(). Purely a beat for pacing -- nothing moves here.
        // ---------------------------------------------------------------
        case GameState::TALK:
            if (stateTimer >= 2.0) { state = GameState::WALK; stateTimer = 0.0; }
            break;

        // ---------------------------------------------------------------
        // WALK: the citizen is released to walk normally along the
        // sidewalk while Spider-Man swings back off-frame (a simple sine
        // bob standing in for a swing motion as he exits stage-left).
        // Once he's fully off-screen the scene resolves to SUCCESS.
        // ---------------------------------------------------------------
        case GameState::WALK: {
            Point2D mv = translatePoint(victim, 30.0*FIXED_DT, 0.0);
            victim = { mv.x, SIDEWALK_Y + 4.0 };
            victimWalkPhase += FIXED_DT * 6.0;
            Point2D ms = translatePoint(spider, -260.0*FIXED_DT, 0.0);
            spider = { ms.x, buildingTopY(WEB_BUILDING) - 10.0 + 40.0*sin(stateTimer*3.0) };
            if (spider.x < -60.0) {
                state = GameState::SUCCESS; stateTimer = 0.0; spiderActive = false;
                logEvent("MISSION SUCCESS -- citizen safely delivered to the walkway.");
            }
            break;
        }

        // ---------------------------------------------------------------
        // SUCCESS / FAILURE: terminal states. The scene keeps animating
        // in the background (traffic, pedestrians, lighting) but nothing
        // in the rescue story advances until the player presses 'R',
        // which calls restartGame() and returns to WAITING.
        // ---------------------------------------------------------------
        case GameState::SUCCESS:
        case GameState::FAILURE:
            break; // wait for R
    }
}

// =====================================================================
// KEYBOARD
// =====================================================================
// Single classic-GLUT keyboard callback (glutKeyboardFunc), registered
// once in main(). Every key is handled case-insensitively by listing both
// the lower- and upper-case ASCII values, since GLUT does not normalize
// case for the standard keyboard callback the way some toolkits do.
void keyboard(unsigned char key, int, int) {
    switch (key) {
        case 's': case 'S': triggerSave(); break;
        case 'd': case 'D': requestLightMode(LightingMode::DAY); break;
        case 'n': case 'N': requestLightMode(LightingMode::NIGHT); break;
        case 'p': case 'P': isPaused = !isPaused; logEvent(isPaused ? "Simulation paused." : "Simulation resumed."); break;
        case 'g': case 'G': showDebugGrid = !showDebugGrid; break;
        case 'h': case 'H': showControlsPanel = !showControlsPanel; break;
        case 'r': case 'R': restartGame(); break;
        case 27: case 'q': case 'Q': exit(0);
    }
}

// =====================================================================
// GLUT CALLBACKS
// =====================================================================
// display() draws exactly one frame in back-to-front order: sky and
// celestial objects first, then background scenery, then the road layer
// and everything on it, then the rescue actors, then all overlay/HUD
// content last so it always sits on top regardless of what's beneath it.
void display() {
    glClear(GL_COLOR_BUFFER_BIT);
    glLoadIdentity();

    // Brief camera shake on failure -- a small, decaying random offset
    // computed once per frame via translatePoint, applied as a GL
    // translation so every primitive drawn this frame inherits it without
    // each draw function needing to know about it.
    if (shakeTimeLeft > 0.0) {
        double mag = 4.0 * (shakeTimeLeft / 0.35);
        Point2D offset = translatePoint({0,0}, ((rand()%200)-100)/100.0*mag, ((rand()%200)-100)/100.0*mag);
        glTranslatef((float)offset.x, (float)offset.y, 0.0f);
    }

    drawSky();
    drawSunMoon();
    drawClouds();
    drawBirds();
    drawPlane();
    drawBuildings();
    drawTrees();
    drawRoad();
    drawWetReflections();
    drawStreetLamps();
    drawCars();
    drawPedestrians();

    drawWeb();
    drawVictim();
    drawSpiderMan();
    drawImpactParticles();

    drawDebugGrid();
    drawHUD();

    glutSwapBuffers();
}

// Recomputes the orthographic projection whenever the window is resized,
// keeping world coordinates fixed at (0,0)-(WINDOW_WIDTH,WINDOW_HEIGHT)
// regardless of the actual pixel size of the window -- so the scene
// stretches to fill the window rather than the coordinate system shifting.
void reshape(int w, int h) {
    glViewport(0, 0, w, h);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, WINDOW_WIDTH, 0, WINDOW_HEIGHT);
    glMatrixMode(GL_MODELVIEW);
}

// The whole simulation advances on this fixed-rate timer rather than in
// the display callback, so game logic runs at a consistent 1/TIMER_MS
// rate independent of how often the window actually gets redrawn.
void timerFunc(int) {
    updateGame();
    glutPostRedisplay();
    glutTimerFunc(TIMER_MS, timerFunc, 0);
}

// =====================================================================
// DIAGNOSTICS
// =====================================================================

// Lightweight console logger for state-changing events (pause/resume,
// rescue attempts, mission outcomes, lighting changes). Timestamps aren't
// necessary here since the tick loop is fixed-rate; this exists purely to
// give a readable trace of what happened during a run for debugging or a
// viva walkthrough, without cluttering the on-screen HUD.
void logEvent(const string& msg) {
    printf("[event] %s\n", msg.c_str());
}

// Sanity-checks the seven-building skyline layout at startup: confirms no
// two buildings overlap horizontally and that the key story buildings
// (jump / anchor / web-launch / landing) are distinct and in a sensible
// left-to-right order. Doesn't stop the program on failure -- it's a
// development aid, printed once before the window opens -- but it catches
// layout mistakes (e.g. two buildings assigned the same index) immediately
// instead of letting them show up as a visually broken scene.
void verifyLayoutNonOverlapping() {
    bool anyOverlap = false;
    for (size_t i = 0; i < buildings.size(); ++i) {
        for (size_t j = i + 1; j < buildings.size(); ++j) {
            double leftI = buildings[i].x, rightI = buildings[i].x + buildings[i].w;
            double leftJ = buildings[j].x, rightJ = buildings[j].x + buildings[j].w;
            bool overlap = leftI < rightJ && leftJ < rightI;
            if (overlap) {
                anyOverlap = true;
                printf("[layout warning] Building %zu overlaps Building %zu\n", i+1, j+1);
            }
        }
    }
    if (!anyOverlap) {
        printf("[layout check] All %zu buildings are clear of one another.\n", buildings.size());
    }

    int keyIndices[4] = { JUMP_BUILDING, ANCHOR_BUILDING, WEB_BUILDING, LAND_BUILDING };
    const char* keyNames[4] = { "jump", "anchor", "web-launch", "landing" };
    bool distinctOk = true;
    for (int a = 0; a < 4 && distinctOk; ++a)
        for (int b = a + 1; b < 4; ++b)
            if (keyIndices[a] == keyIndices[b]) {
                printf("[layout warning] %s and %s buildings share the same index (%d)\n",
                       keyNames[a], keyNames[b], keyIndices[a]);
                distinctOk = false;
            }
    if (distinctOk) {
        printf("[layout check] jump/anchor/web-launch/landing buildings are all distinct "
               "(Building %d / %d / %d / %d).\n",
               JUMP_BUILDING+1, ANCHOR_BUILDING+1, WEB_BUILDING+1, LAND_BUILDING+1);
    }
}

void printStartupBanner() {
    printf("=========================================================\n");
    printf("  SPIDER-MAN RESCUE -- CG Lab Project\n");
    printf("  Controls: S save | D day | N night | P pause | G grid\n");
    printf("            R restart | Esc/Q quit\n");
    printf("=========================================================\n");
}

// =====================================================================
// RESTART / INIT
// =====================================================================
// Resets every piece of mutable rescue-story state back to its starting
// values and re-seeds traffic/pedestrians -- called both once at startup
// (from initGL) and every time the player presses 'R'. Deliberately does
// NOT reset lighting mode, tick counter, or the debug/help panel toggles,
// since those are session-level settings the player likely wants to
// persist across a restart rather than snapping back to defaults.
void restartGame() {
    state = GameState::WAITING;
    stateTimer = 0.0;
    countdown = COUNTDOWN_START;
    isPaused = false;

    victim = { buildingCenterX(JUMP_BUILDING), buildingTopY(JUMP_BUILDING) };
    victimVel = {0,0};
    victimWalkPhase = 0.0;

    spider = { buildingCenterX(WEB_BUILDING), buildingTopY(WEB_BUILDING) };
    spiderActive = false;
    showWeb = false;

    particlesActive = false;
    particles.clear();

    initVehicles();
    initPedestrians();
    logEvent("Scene restarted.");
}

// One-time OpenGL and scene setup, called once from main() before the
// GLUT callbacks are registered. Blending and point/line smoothing are
// enabled globally up front since every primitive in the scene (particles,
// glow rings, antialiased strokes) relies on them; there's no per-draw
// state toggling needed anywhere else in the program as a result.
void initGL() {
    glClearColor(0, 0, 0, 1);
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
    glEnable(GL_POINT_SMOOTH);
    glEnable(GL_LINE_SMOOTH);
    glPointSize(1.6f);
    glLineWidth(1.4f);

    printStartupBanner();

    initBuildings();
    initEnvironment();
    initBirds();
    currentPalette = paletteForMode(activeLightMode);
    verifyLayoutNonOverlapping();
    restartGame();
}

// =====================================================================
// MAIN
// =====================================================================
// Standard classic-GLUT bootstrap: create the window, run one-time GL and
// scene initialization, register the three callbacks the whole program
// needs (display, reshape, keyboard), start the fixed-rate timer that
// drives every animation and state transition, then hand control to
// glutMainLoop(), which never returns during normal operation -- the
// program exits via the Esc/Q key calling exit(0) directly.
int main(int argc, char** argv) {
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGBA);
    glutInitWindowSize(WINDOW_WIDTH, WINDOW_HEIGHT);
    glutInitWindowPosition(60, 40);
    glutCreateWindow(WINDOW_TITLE);

    initGL();

    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutKeyboardFunc(keyboard);
    glutTimerFunc(TIMER_MS, timerFunc, 0);

    glutMainLoop();
    return 0;
}
